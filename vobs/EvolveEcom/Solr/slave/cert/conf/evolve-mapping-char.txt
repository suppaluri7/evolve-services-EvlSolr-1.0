# Using this file to remove or replace special character such as the trademark registered character.
#
# While upgrading from Solr3 to Solr8, we see difference in the way StandardTokenizerFactory
# remove special character. Therefore, this file has been used to remove/replace special character
# that we do not want index and /or search.
#
# Syntax:
#   "source" => "target"
#     "source".length() > 0 (source cannot be empty.)
#     "target".length() >= 0 (target can be empty.)
# Ⓡ  [CIRCLED LATIN CAPITAL LETTER R]
"\u00AE" => ""
# ™  [TRADE MARK]
"\u2122" => ""
# ©  [COPYRIGHT]
"\u00A9" => ""
# ℠ [SERVICE MARK]
"\u2120" => ""
# ℗  [SOUND RECORDING COPYRIGHT]
"\u2117" => ""
